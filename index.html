import streamlit as st import sympy as sp from sympy.parsing.sympy_parser import parse_expr, standard_transformations, implicit_multiplication_application, convert_xor import pytesseract from PIL import Image import re # ========== কনফিগারেশন ========== # Tesseract-এর পাথ (উইন্ডোজে দরকার; ম্যাক/লিনাক্সে কমেন্ট করুন) # pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files\Tesseract-OCR\tesseract.exe' st.set_page_config(page_title="সর্বজনীন গণিত সমাধানকারী", page_icon="🧮", layout="centered") # ========== হেলপার ফাংশন ========== def preprocess_math_expression(expr_str: str) -> str: """ ব্যবহারকারীর টেক্সটকে SymPy-উপযোগী স্ট্রিং-এ রূপান্তর করে। যেমন: 8 ÷ 2(2+2) -> 8/(2*(2+2)) """ # ÷ এবং : চিহ্নকে / তে রূপান্তর expr_str = expr_str.replace('÷', '/').replace(':', '/') # ইমপ্লিসিট গুণ (যেমন 2(2+2) -> 2*(2+2)) expr_str = re.sub(r'(\d)(\()', r'\1*\2', expr_str) expr_str = re.sub(r'(\))(\d)', r'\1*\2', expr_str) expr_str = re.sub(r'(\))(\()', r'\1*\2', expr_str) # ^ কে ** তে রূপান্তর (পাওয়ার) expr_str = expr_str.replace('^', '**') # x² এর মতো ক্ষেত্রে (ইউনিকোড সুপারস্ক্রিপ্ট) হ্যান্ডলিং sup_map = {'⁰': '0', '¹': '1', '²': '2', '³': '3', '⁴': '4', '⁵': '5', '⁶': '6', '⁷': '7', '⁸': '8', '⁹': '9'} for sup, digit in sup_map.items(): expr_str = expr_str.replace(sup, f'**{digit}') return expr_str def solve_math(problem: str) -> str: """ যেকোনো গাণিতিক সমস্যা SymPy দিয়ে সমাধান করে। রিটার্ন করে সুন্দর ফরম্যাটেড উত্তর। """ problem = problem.strip() if not problem: return "⚠️ দয়া করে একটি গাণিতিক সমস্যা লিখুন।" # প্রি-প্রসেসিং try: expr_str = preprocess_math_expression(problem) except Exception as e: return f"❌ প্রি-প্রসেসিং ব্যর্থ: {str(e)}" # SymPy ট্রান্সফরমেশন (ইমপ্লিসিট গুণের জন্য) transformations = standard_transformations + (implicit_multiplication_application, convert_xor) # সমস্যার ধরন শনাক্ত করে উপযুক্ত অপারেশন try: # 1) সমীকরণ কিনা (একটি '=' চিহ্ন আছে?) if '=' in expr_str and not expr_str.startswith('=='): # সমীকরণ সমাধান left, right = expr_str.split('=', 1) lhs = parse_expr(left, transformations=transformations) rhs = parse_expr(right, transformations=transformations) eqn = sp.Eq(lhs, rhs) # চলক (ভেরিয়েবল) বের করি vars = list(eqn.free_symbols) if not vars: return f"🔍 সমীকরণে কোনো চলক নেই। ধ্রুবক: {sp.simplify(eqn)}" sol = sp.solve(eqn, vars[0]) if sol: return f"✅ সমাধান: {vars[0]} = {sol}" else: return "⚠️ সমাধান পাওয়া যায়নি (সমীকরণটি অমীমাংসিত অথবা কোনও বাস্তব সমাধান নেই)।" # 2) ডেরিভেটিভ (derivative, d/dx, ' ইত্যাদি) elif 'derivative' in problem.lower() or "diff" in problem.lower() or "ডেরিভ" in problem: # ফাংশন এবং চলক বের করার চেষ্টা # সহজ পদ্ধতি: ব্যবহারকারী যদি "derivative of x^2" টাইপ করে, তাহলে আমরা extract করি match = re.search(r'derivative of (.+)', problem.lower()) if match: func_str = match.group(1) else: func_str = expr_str expr = parse_expr(func_str, transformations=transformations) var = list(expr.free_symbols)[0] if expr.free_symbols else sp.Symbol('x') deriv = sp.diff(expr, var) return f"📈 ডেরিভেটিভ (d/d{var}): {sp.latex(deriv)}" # 3) ইন্টিগ্রাল (integral, ∫) elif 'integral' in problem.lower() or '∫' in problem: # ∫₀¹ x² dx এর মতো পার্স করা জটিল; ব্যবহারকারীকে সঠিক লিখতে বলি বা সরল করুন # সহজ নিয়ম: "integrate x^2 from 0 to 1" অথবা "∫ x^2 dx" if '∫' in problem: # সরলীকৃত: ইউনিকোড ইন্টিগ্রাল চিহ্ন থাকলে স্ট্রিং ক্লিন problem = problem.replace('∫', 'integrate') if 'integrate' in problem.lower(): # স্ট্রিং থেকে ফাংশন এবং সীমা বের করার চেষ্টা (খুব জটিল) # সংক্ষেপে: যদি 'from' এবং 'to' থাকে, তাহলে definite integral match = re.search(r'integrate (.+?) (?:from (.+?) to (.+))?', problem.lower()) if match: func_str = match.group(1) expr = parse_expr(func_str, transformations=transformations) if match.group(2) and match.group(3): a = parse_expr(match.group(2)) b = parse_expr(match.group(3)) integ = sp.integrate(expr, (sp.Symbol('x'), a, b)) return f"∫ নির্দিষ্ট ইন্টিগ্রাল: {sp.latex(integ)}" else: integ = sp.integrate(expr, sp.Symbol('x')) return f"∫ অনির্দিষ্ট ইন্টিগ্রাল: {sp.latex(integ)} + C" # 4) সাধারণ এক্সপ্রেশন (সরলীকরণ, মূল্যায়ন) else: expr = parse_expr(expr_str, transformations=transformations) # সংখ্যাসূচক মান বের করার চেষ্টা (যদি সব চলক সংখ্যা হয়) try: numerical = float(sp.N(expr)) # যদি পূর্ণসংখ্যা হয়, int দেখান if numerical.is_integer(): numerical = int(numerical) simplified = sp.simplify(expr) return f"🔢 সরলীকৃত ফলাফল: {sp.latex(simplified)}\n\n📊 সংখ্যাসূচক মান: {numerical}" except: simplified = sp.simplify(expr) return f"🔢 সরলীকৃত রাশি: {sp.latex(simplified)}" except sp.SympifyError as e: return f"❌ SymPy পার্সিং ত্রুটি: {str(e)}\n💡 টিপ: '2+2', 'x**2 + 3*x', 'integrate x**2' ইত্যাদি ব্যবহার করুন।" except Exception as e: return f"❌ অজানা ত্রুটি: {str(e)}" def extract_text_from_image(image_file) -> str: """ আপলোড করা ছবি থেকে OCR-এর মাধ্যমে টেক্সট বের করে। """ try: img = Image.open(image_file) # ছবির সাইজ বাড়ালে OCR ভালো হয় img = img.resize((img.width*2, img.height*2)) text = pytesseract.image_to_string(img, lang='eng+ben') # ইংরেজি + বাংলা # ম্যাথ এক্সপ্রেশন পরিষ্কার করা text = re.sub(r'[^\w\d\s\+\-\*/\(\)\=\^\^\%\√\[\]\{\}\<\>\.\,\!\?\;\:\|\\]', '', text) return text.strip() except Exception as e: return f"OCR ত্রুটি: {str(e)}" # ========== UI ডিজাইন ========== st.title("🧠 সর্বজনীন গণিত সমাধানকারী") st.markdown("টেক্সট টাইপ করে অথবা ছবি আপলোড করে যেকোনো গাণিতিক সমস্যার সমাধান পেতে পারেন।") # ট্যাব তৈরি: টেক্সট ইনপুট ও ইমেজ ইনপুট tab1, tab2 = st.tabs(["✍️ টেক্সট ইনপুট", "🖼️ ইমেজ আপলোড"]) with tab1: st.subheader("ম্যানুয়ালি সমস্যা লিখুন") # প্রি-ডিফাইন্ড উদাহরণ বাটন examples = [ "8 ÷ 2(2+2)", "2x + 5 = 13", "∫ x² dx", "derivative of 3x³-2x²+7", "Area of circle radius 7", "x²-9=0", "Pythagorean theorem a=3, b=4", "Factor x²+5x+6", "Solve 2x²+5x-3=0", "∫ sin(x) cos(x) dx", "lim (x→0) sin(x)/x", "Matrix [[1,2],[3,4]] inverse" ] cols = st.columns(3) for i, ex in enumerate(examples): if cols[i%3].button(f"📌 {ex}", key=f"ex_{i}"): st.session_state.text_input = ex # টেক্সট ইনপুট ফিল্ড user_input = st.text_area("আপনার গাণিতিক সমস্যা লিখুন:", value=st.session_state.get("text_input", ""), height=100) if st.button("🔍 সমাধান করুন (টেক্সট)", type="primary"): if user_input: with st.spinner("গণনা চলছে..."): answer = solve_math(user_input) st.success("✅ ফলাফল:") st.latex(answer) if answer.startswith("$") else st.write(answer) else: st.warning("দয়া করে একটি সমস্যা লিখুন।") with tab2: st.subheader("ছবি আপলোড করুন") uploaded_file = st.file_uploader("গাণিতিক সমস্যার ছবি নির্বাচন করুন (JPG, PNG)", type=["jpg", "jpeg", "png"]) if uploaded_file is not None: col1, col2 = st.columns(2) with col1: st.image(uploaded_file, caption="আপলোড করা ছবি", width=250) if st.button("🔍 ছবি থেকে সমাধান করুন", type="primary"): with st.spinner("ছবি থেকে টেক্সট বের করা হচ্ছে..."): extracted_text = extract_text_from_image(uploaded_file) if extracted_text and not extracted_text.startswith("OCR ত্রুটি"): st.info(f"📝 ছবি থেকে প্রাপ্ত টেক্সট: `{extracted_text}`") with st.spinner("গাণিতিক সমাধান চলছে..."): answer = solve_math(extracted_text) st.success("✅ ফলাফল:") st.write(answer) else: st.error(f"ছবি থেকে টেক্সট বের করতে ব্যর্থ।\n{extracted_text}") st.markdown("---") st.markdown("### 📚 উপলব্ধ গণিতের ধরণসমূহ:") st.markdown(""" - **পাটিগণিত** : `2 + 3 * 4`, `8 ÷ 2(2+2)` - **বীজগণিত** : `x^2 - 5x + 6 = 0`, `Factor x^2 + 5x + 6` - **ক্যালকুলাস** : `derivative of x^3`, `integral x^2 dx`, `∫₀¹ x² dx` (লিখতে পারেন `integrate x^2 from 0 to 1`) - **জ্যামিতি** : `Area of circle radius 7` (SymPy সরাসরি বাক্য বুঝে না, তাই এগুলোর জন্য `π*7**2` লিখুন) - **ম্যাট্রিক্স** : `Matrix([[1,2],[3,4]]).inv()` - **লিমিট** : `limit(sin(x)/x, x, 0)` """) st.markdown("> **দ্রষ্টব্য:** 'Area of circle' বা 'Pythagorean theorem' এর মতো প্রাকৃতিক ভাষা পুরোপুরি বোঝে না। দয়া করে সূত্র সরাসরি লিখুন, যেমন `π*7**2` অথবা `sqrt(3**2+4**2)`।") #
